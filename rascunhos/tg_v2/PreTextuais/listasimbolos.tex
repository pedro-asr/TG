\begin{listofsymbols}
\item[$G$]          Irradiância solar [\si{\watt\per\square\meter}]
\item[$G_h$]        Irradiância horizontal global (GHI) [\si{\kWh\per\square\metre\per\day}]
\item[$G_t$]        Irradiância no plano dos módulos [\si{\watt\per\square\meter}]
\item[$G_0$]        Irradiância extraterrestre no plano horizontal [\si{\watt\per\square\meter}]
\item[$k_t$]        Índice de claridade ($G_h / G_0$)
\item[$k_d$]        Fração difusa ($G_d / G_h$)
\item[$P_{ac}$]     Potência CA gerada [\si{\watt}]
\item[$P_{dc}$]     Potência CC gerada [\si{\watt}]
\item[$P_{\mathrm{nom}}$] Potência nominal da usina [\si{\watt\peak}]
\item[$T_{amb}$]    Temperatura ambiente [\si{\celsius}]
\item[$T_{mod}$]    Temperatura do módulo fotovoltaico [\si{\celsius}]
\item[$\eta$]       Eficiência do sistema fotovoltaico
\item[$\gamma$]     Coeficiente de temperatura de potência [\si{\per\celsius}]
\item[$\hat{y}$]    Valor previsto pelo modelo
\item[$y$]          Valor real observado
\item[$h$]          Horizonte de previsão
\end{listofsymbols}
